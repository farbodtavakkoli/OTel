# Shared client environment for inference/llamacpp — llama.cpp llama-server serving
# a Qwen3.8-27B GGUF LLM, EmbeddingGemma-300M embeddings, and Qwen3-Reranker-0.6B.
#
# llama.cpp is a C++/HIP/CUDA build, not a pip package. Build it once (one build serves all
# three leaves), then install these client deps into one venv at this root (Python 3.12):
#   python3 -m venv .env_llamacpp && .env_llamacpp/bin/pip install -r requirements.txt
#
# Build -- AMD / ROCm (gfx950); needs cmake, ninja, g++, libssl-dev and ROCm at /opt/rocm:
#   git clone --depth 1 https://github.com/ggml-org/llama.cpp && cd llama.cpp
#   cmake -S . -B build -G Ninja -DGGML_HIP=ON -DGPU_TARGETS=gfx950 \
#     -DCMAKE_BUILD_TYPE=Release -DLLAMA_OPENSSL=ON
#   cmake --build build --config Release -j 32
# Build -- NVIDIA / CUDA (Hopper cc 9.0); same, only the backend flag changes (no arch flag
# needed, cmake detects 90-real):
#   export PATH=/usr/local/cuda/bin:$PATH CUDACXX=/usr/local/cuda/bin/nvcc
#   cmake -S . -B build -G Ninja -DGGML_CUDA=ON -DCMAKE_BUILD_TYPE=Release \
#     -DLLAMA_OPENSSL=ON -DLLAMA_CURL=OFF     # -DLLAMA_CURL=OFF optional: forces cpp-httplib
#   cmake --build build --config Release -j 32
#
# -DLLAMA_OPENSSL=ON + libssl-dev are required on both backends for the `-hf` downloader:
# without them every pull fails with "HTTPS is not supported" plus a misleading
# "failed to load model, ''". (-DLLAMA_BUILD_BORINGSSL=ON / -DLLAMA_BUILD_LIBRESSL=ON also work.)
# If cmake/ninja are missing, pip-install them into a throwaway venv; if the model share is
# NFS and rejects renames, put the build tree and LLAMA_CACHE on tmpfs (/dev/shm).
requests==2.34.2
