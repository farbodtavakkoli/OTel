# Python client dependencies for inference/tei (host Python 3.12) -- HF Text Embeddings
# Inference serving google/embeddinggemma-300m.
#
#   python3 -m venv .env_tei          # at inference/tei/
#   source .env_tei/bin/activate
#   pip install -r requirements.txt
#
# NOTE: the venv fails on a CIFS/NFS share (pip's bootstrap cannot write symlinks there) --
# build it on local or tmpfs disk instead.
#
# TEI's SERVER is a Rust binary, not a pip package. Install it per-vendor:
#
# NVIDIA / CUDA -- prebuilt, arch-specific image, no source build. For H100 (Hopper, compute
# capability 9.0) use the `hopper` tag; the generic `cuda-*` tag is the wrong arch.
#   unset HTTP_PROXY HTTPS_PROXY ALL_PROXY http_proxy https_proxy all_proxy   # ghcr.io is
#   sudo docker pull ghcr.io/huggingface/text-embeddings-inference:hopper-1.9 # often proxied
#   sudo docker run -d --name tei_cuda --gpus '"device=0"' -p 8090:80 \
#     -v $HF_HOME:/data -e HF_HOME=/data -e HF_TOKEN=<from dev.env> \
#     ghcr.io/huggingface/text-embeddings-inference:hopper-1.9 \
#     --model-id google/embeddinggemma-300m --dtype float32 --port 80
#   TEI listens on :80 in-container; HF_HOME=/data means the cache is read from /data/hub.
#   --dtype float32 is required: the Hopper image rejects bfloat16, and the EmbeddingGemma
#   card warns against fp16. Generic non-Hopper arches use the :cuda-1.9 tag the same way.
#
# AMD / ROCm -- there is no published AMD image, so the router must be built from source
# against a ROCm PyTorch base (e.g. lmsysorg/sglang-rocm:v0.5.17-rocm720-mi35x-20260819,
# which carries ROCm 7.2 torch for gfx950; upstream's guide uses rocm/pytorch:latest). Rust
# and protoc are not in the base image and must be added. Inside the container:
#   pip install --no-deps -r backends/python/server/requirements-amd.txt
#   cargo build --release --bin text-embeddings-router -F python -F http --no-default-features
#   ./target/release/text-embeddings-router \
#       --model-id google/embeddinggemma-300m --dtype bfloat16 --port 8301
#   No source patches are needed for gfx950. --dtype bfloat16 is deliberate; do not force FP16.
#   Do not install the ROCm python-backend requirements into the client venv -- they are
#   container-only and pin `python_version < "3.13"`. Torch comes from the base image.
#
# TEI serves embeddings / sequence-classification only: there is no inference/tei/llm and no
# inference/tei/reranker. Use inference/llamacpp/reranker or inference/vllm/reranker.
