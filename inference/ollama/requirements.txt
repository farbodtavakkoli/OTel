# Requirements for inference/ollama — Ollama serving GGUF models on ROCm
# (llm/: Qwen3.8-27B Q8_0; embedding/: EmbeddingGemma-300M Q8_0).
#
# Ollama is a single static Go binary shipped inside a container, NOT a pip package:
# there is no `pip install ollama-server`. This file records (a) the exact CONTAINER +
# docker flags that were verified, (b) the Modelfiles that register the already-cached
# GGUFs, and (c) the small python client dependencies shared by both leaf clients.
#
# VENV CONVENTION: python3 -m venv .env_ollama  (at inference/ollama/).
#
# ---------------------------------------------------------------------------
# (a) CONTAINER IMAGE — verified on 8x AMD Instinct MI355X (gfx950)
# ---------------------------------------------------------------------------
# Image:     ollama/ollama:rocm
# Digest:    sha256:b5d0813b23540019b2a40d70af6531b1d76e5ebcbf17899e5922445c627b6063
# Image ID:  b5d0813b2354
# On-disk:   4.55 GB uncompressed (1.43 GB compressed pull)
# Ollama:    0.32.14   (`docker exec <name> ollama --version`)
# Backend:   library=ROCm compute=gfx950 libdirs=ollama,rocm_v7_2
# Host:      ROCm 7.2.4, Ubuntu, Docker 29.7.2, Python 3.12.3
#
# The image bundles its OWN ROCm userspace (rocm_v7_2). The host only supplies the
# amdgpu KFD driver, so no ROCm install or `HSA_OVERRIDE_GFX_VERSION` is needed and
# gfx950 is detected natively. One daemon can hold both models at once (multi-model
# appliance); a dedicated per-workload container is equally valid.
#
#   docker pull ollama/ollama:rocm
#
# EXACT run command that worked (single GPU — physical GPU 2 only). $DATA_DIR is the
# Ollama model store, $LLAMA_CACHE the existing GGUF cache (both set in README.md):
#   docker run -d \
#     --device /dev/kfd \
#     --device /dev/dri/renderD144 \
#     -v $DATA_DIR/ollama:/root/.ollama \
#     -v $LLAMA_CACHE:/ggufs:ro \
#     -p 11434:11434 \
#     --name ollama_llm \
#     ollama/ollama:rocm
# Multi-GPU (physical GPUs 2 AND 3): add --device /dev/dri/renderD152.
# Embedding scale-out: second container on the other render node, -p 11435:11434.
#
# FLAG NOTES:
#   --device /dev/kfd                 mandatory; the ROCm compute node.
#   --device /dev/dri/renderDNNN      per-GPU pinning. Passing the whole /dev/dri (as the
#                                     canonical run line does) would expose ALL 8 GPUs. Map:
#                                       rocm-smi --showbus            # GPU[n] -> PCI addr
#                                       ls -l /dev/dri/by-path/       # PCI addr -> renderDNNN
#                                     Here: GPU2=0000:a5:00.0=renderD144
#                                           GPU3=0000:dc:00.0=renderD152
#   -v $DATA_DIR/ollama:/root/.ollama
#                                     model store. Do NOT use the `-v ollama:...`
#                                     named volume: that lands under /var/lib/docker on the
#                                     root filesystem, and `ollama create` COPIES blobs in.
#                                     The store measured 55 GB after both models.
#   -v $LLAMA_CACHE:/ggufs:ro
#                                     read-only view of the already-downloaded GGUFs so
#                                     Modelfiles register them without a new pull. HF
#                                     snapshot symlinks (../../blobs/<sha>) resolve
#                                     because the whole tree is mounted.
#   -p 11434:11434                    Ollama's default port; used as-is.
#
# Useful environment overrides (pass with -e):
#   -e OLLAMA_SCHED_SPREAD=1          force a model across every visible GPU
#   -e OLLAMA_CONTEXT_LENGTH=8192     cap context; the default is derived from TOTAL VRAM
#   -e OLLAMA_NUM_PARALLEL=4          concurrent request slots (raise for batch embedding)
#   -e OLLAMA_KEEP_ALIVE=30m          keep models resident longer than the 5m default
#   -e OLLAMA_MAX_LOADED_MODELS=2     allow the LLM and the embedder to coexist
#
# NVIDIA variant — VERIFIED on 8x NVIDIA H100 80GB HBM3 (single GPU: physical
# GPU 4 only; shared node). Per repo convention only the image tag + device flags change;
# every Modelfile/API command is identical, and that held exactly. See README.md "H100" §.
#   Image:    ollama/ollama  (Docker Hub :latest)
#   Digest:   sha256:57d60e686821ea81a7748a3ec8141308c8b8f95b27105713954abf7a6529e700
#   On-disk:  8.43 GB
#   Ollama:   0.32.15   (`docker exec ollama_h100 ollama --version`)
#   Backend:  library=CUDA compute=9.0 libdirs=ollama,cuda_v13 driver=13.0  (native Hopper,
#             CUDA-13 userspace bundled in the image; host supplies only the NVIDIA driver —
#             no build, no patch, no HSA_OVERRIDE analogue needed)
#   Host:     driver 580.173.02, CUDA 13.0, Docker, Python 3.12.3
#
#   If Docker Hub is proxy-blocked, pull with the proxy unset:
#     unset HTTP_PROXY HTTPS_PROXY ALL_PROXY http_proxy https_proxy all_proxy
#     sudo docker pull ollama/ollama
#
#   EXACT run command that worked (single GPU — physical GPU 4 only, non-default port 11440):
#     sudo docker run -d --gpus '"device=4"' \
#       -e OLLAMA_HOST=0.0.0.0:11440 \
#       -v $GGUF:/ggufs:ro \
#       -v $STORE:/root/.ollama \
#       -p 127.0.0.1:11440:11440 \
#       --name ollama_h100 ollama/ollama
#   FLAG NOTES (deviations from the ROCm run line, all H100/shared-node driven):
#     --gpus '"device=4"'   replaces --device /dev/kfd + --device /dev/dri/renderD*. This
#                           alone confines Ollama to ONE card (discovery then lists a single
#                           CUDA0=H100). GPUs 0-3 were a co-tenant production job — untouched.
#     OLLAMA_HOST/-p 11440  non-default port to avoid collisions with co-tenant agents.
#     -v $STORE:/root/.ollama   put the store on /dev/shm tmpfs (ollama create COPIES blobs;
#                           keep off /). tmpfs also dodged an "Operation not permitted"
#                           atomic-rename failure some network filesystems throw at pip/venv.
#   GPU-residency PROVEN: `ollama ps` -> PROCESSOR "100% GPU" for BOTH models at once; runner
#   offload log "offloaded N/N layers to GPU" (all CUDA0 buffers); nvidia-smi -i 4 shows the
#   /usr/lib/ollama/llama-server PIDs holding VRAM (LLM 5942 MiB + embedder 1030 MiB) on GPU 4.
#
#   GGUFs — if the GGUF cache is empty (no *-GGUF repos in HF_HOME), the (b) FROM paths
#   below resolve to nothing. Download first (proxy unset), then point the Modelfile FROM
#   at the local path. SMALL GGUFs give a fast smoke (the 29 GB 27B also fits 80 GB):
#     hf download unsloth/Qwen3-0.6B-GGUF Qwen3-0.6B-Q8_0.gguf --local-dir <dir>         # 639 MB
#     hf download ggml-org/embeddinggemma-300M-GGUF embeddinggemma-300M-Q8_0.gguf --local-dir <dir>  # 334 MB, same repo as (b)
#   H100 Modelfile variants (leave the MI355X Modelfiles untouched):
#     llm:  FROM /ggufs/Qwen3-0.6B-GGUF/Qwen3-0.6B-Q8_0.gguf                    -> ollama create qwen3-0.6b-q8
#     emb:  FROM /ggufs/embeddinggemma-300M-GGUF/embeddinggemma-300M-Q8_0.gguf  -> ollama create embeddinggemma
#   The embedding layer sha b5ce9d77a3fc… matches (b) byte-for-byte — same model, just fetched.
#
# The python client deps below are hardware-neutral and unchanged on H100 (requests 2.34.2,
# python-dotenv 1.2.3 install and run identically; the clients only speak JSON over HTTP).
#
# ---------------------------------------------------------------------------
# (b) MODELFILES — register cached GGUFs, avoiding re-downloads
# ---------------------------------------------------------------------------
# llm/Modelfile.qwen3-27b:
#   FROM /ggufs/models--unsloth--Qwen3.8-27B-GGUF/snapshots/27af057ecb382ddfea5d12837360a8980560e3ed/Qwen3.8-27B-Q8_0.gguf
#   docker cp llm/Modelfile.qwen3-27b ollama_llm:/root/Modelfile.qwen3-27b
#   docker exec ollama_llm ollama create qwen3.8-27b-q8 -f /root/Modelfile.qwen3-27b
#   Measured: 1 m 41 s (copies the 29 GB blob into the store). Source:
#   unsloth/Qwen3.8-27B-GGUF, Qwen3.8-27B-Q8_0.gguf, 29047086048 bytes,
#   sha256 a680f44a06920e5d689774823782006aa3acc8db95750323373b24139b67e348
#
# embedding/Modelfile.embeddinggemma:
#   FROM /ggufs/models--ggml-org--embeddinggemma-300M-GGUF/snapshots/0f741b5a6585bd53aeb15cd1372c56f2a0f65e12/embeddinggemma-300M-Q8_0.gguf
#   docker cp embedding/Modelfile.embeddinggemma ollama_embed:/root/Modelfile.embeddinggemma
#   docker exec ollama_embed ollama create embeddinggemma -f /root/Modelfile.embeddinggemma
#   Measured: 1.4 s. Source: ggml-org/embeddinggemma-300M-GGUF (319 MB),
#   sha256 b5ce9d77a3fc4b3b39ccb5643c36777911cc4eb46a66962eadfa3f5f60490d63
#   NOTE — no TEMPLATE directive is set, deliberately. EmbeddingGemma's task prefixes
#   are applied CLIENT-SIDE by embedding/inference_embedding_ollama.py, because Ollama's
#   /api/embed sends input through verbatim. See embedding/README.md.
#
# Paths in the FROM lines are CONTAINER paths (the :ro bind mount above).
#
# ---------------------------------------------------------------------------
# (c) PYTHON CLIENT DEPENDENCIES — shared by both leaf clients
# ---------------------------------------------------------------------------
# Python 3.12.3. Install:
#   python3 -m venv .env_ollama
#   .env_ollama/bin/pip install -r requirements.txt
#
# The clients are deliberately dependency-light: Ollama speaks plain JSON over HTTP on
# both its native API and the OpenAI-compatible shim, so `requests` is enough — no
# openai/torch/transformers needed. Cosine math in the embedding client uses the stdlib
# `math` module.
requests==2.34.2
python-dotenv==1.2.3
