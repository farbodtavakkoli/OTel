# PATH C-CUDA-H100: NVIDIA H100 80GB / CUDA 13.0 / driver 580.173.02 -- TESTED, WORKS WITH CHANGES.
#   Verified 2026-08-23 on 1x H100 80GB, Python 3.12.3. Single-GPU SFT smoke, flash_attention_2
#   active. Full write-up, run command and log evidence: readme_llmfoundry.md section 8b.
#
#   Resolved versions: torch 2.7.0+cu126, mosaicml (Composer) 0.32.1, llm-foundry 0.23.0.dev0
#   (git 0cdb2f4), transformers 4.51.3, mosaicml-streaming 0.12.0, datasets 3.6.0, numpy 2.1.3,
#   accelerate 1.8.1, flash-attn 2.7.4.post1, triton 3.3.0.
#
#   UNLIKE ROCm, no torch pin-fight: Foundry's `torch>=2.7.0,<2.7.1` resolves to a real CUDA
#   wheel (torch 2.7.0+cu126) that RUNS on this cu13/driver-580 box. `pip install torch` alone
#   gives 2.13.0+cu130; the Foundry deps then DOWNGRADE it to 2.7.0+cu126 -- expected, keep it.
#   TWO Python-3.12 packaging traps: (a) install Foundry from the GIT CHECKOUT, not the PyPI
#   wheel -- the wheel (<=0.22.0) pulls `pathtools`, which imports the removed `imp` module and
#   fails to build; (b) add `setuptools<81` so `pkg_resources` exists (Foundry imports it).
#
#   cd training/llm/llmfoundry
#   python3 -m venv .env_llmfoundry && source .env_llmfoundry/bin/activate
#   export CUDA_VISIBLE_DEVICES=7          # plain CUDA; do NOT set HIP_VISIBLE_DEVICES on NVIDIA
#   pip install -U pip 'setuptools<81' wheel
#   pip install torch numpy               # base check: torch 2.13.0+cu130, bf16 matmul OK on GPU
#
#   # 1. Foundry 0.23.0.dev0 runtime closure by name (torch pin resolves to 2.7.0+cu126).
#   #    Drop mosaicml's [wandb] extra (ancient wandb -> pathtools -> imp); pin modern wandb.
#   pip install \
#     'mosaicml[libcloud,oci,gcs,mlflow]>=0.32.1,<0.33' 'wandb>=0.18,<0.19' \
#     'mlflow>=2.14.1,<3.0' 'accelerate>=0.25,<1.9' 'transformers>=4.51.0,<4.52' \
#     'mosaicml-streaming>=0.12.0,<0.13' 'torch>=2.7.0,<2.7.1' 'datasets>=3.3.2,<3.7' \
#     fsspec==2023.6.0 sentencepiece==0.2.0 einops==0.8.1 'omegaconf>=2.2.3,<3' 'slack-sdk<4' \
#     'mosaicml-cli>=0.6.10,<1' onnx==1.18.0 onnxruntime==1.22.0 'boto3>=1.21.45,<2' \
#     'huggingface-hub[hf_xet]>=0.30.0,<0.34' 'beautifulsoup4>=4.12.2,<5' 'tenacity>=8.2.3,<10' \
#     'catalogue>=2,<3' 'typer<1' GitPython==3.1.44 'python-dotenv>=1.0.1'
#
#   # 2. the checkout (supplies scripts/train/train.py), --no-deps => protects resolved torch
#   cd .env_llmfoundry && git clone --depth 1 https://github.com/mosaicml/llm-foundry.git
#   cd llm-foundry && pip install --no-deps -e . && cd ../..
#
#   # 3. flash-attn: the source build fails ONLY if PIP_CACHE_DIR is on tmpfs (Invalid
#   #    cross-device link when caching the built wheel). Grab the prebuilt wheel directly:
#   curl -sSL -o /tmp/fa.whl \
#     https://github.com/Dao-AILab/flash-attention/releases/download/v2.7.4.post1/flash_attn-2.7.4.post1+cu12torch2.7cxx11abiTRUE-cp312-cp312-linux_x86_64.whl
#   pip install --no-deps /tmp/fa.whl
#   #  (Equivalently: on the checkout, `pip install -e ".[gpu]"` pulls flash-attn==2.7.4.post1
#   #   via the [gpu] extra -- use --no-build-isolation and keep PIP_CACHE_DIR off tmpfs.)
#
#   # 4. RE-VERIFY torch: python -c "import torch;print(torch.__version__, torch.version.cuda)"
#   #    -> 2.7.0+cu126 12.6   ;  `pip check` -> No broken requirements found.
#
#   MODEL DEVIATION: LiquidAI/LFM2.5-350M is model_type `lfm2`, which transformers 4.51.3
#   (Foundry's <4.52 ceiling) does NOT recognize -- it cannot load. Use a qwen3/llama-family
#   chat checkpoint instead (the smoke used Qwen/Qwen3-0.6B). Do NOT use a *reranker* checkpoint
#   (e.g. Qwen3-Reranker-0.6B): its chat template is a yes/no judge that discards user/assistant
#   turns, so every SFT row renders empty and the loader drops them all.
#   Keep the YAML's `attn_implementation: flash_attention_2` on H100 (it builds and runs);
#   only ROCm needs the `sdpa` override.

# PATH C-ROCm7.2: AMD MI355X (gfx950) / ROCm 7.2 -- TESTED, WORKS WITH CHANGES.
#   Verified 2026-08 on 1x MI355X, ROCm 7.2.4, Python 3.12.3. Full write-up, run
#   command and log evidence: readme_llmfoundry.md section 8a.
#
#   Resolved versions: torch 2.11.0+rocm7.2, torchvision 0.26.0+rocm7.2,
#   torchaudio 2.11.0+rocm7.2, mosaicml (Composer) 0.32.1, llm-foundry 0.23.0.dev0,
#   transformers 4.51.3, mosaicml-streaming 0.12.0, datasets 3.6.0, numpy 2.1.3.
#
#   THE CONFLICT: llm-foundry setup.py pins `torch>=2.7.0,<2.7.1` and Composer 0.32.1
#   (the newest release) pins `torch<2.7.1,>=2.6.0` + `torchvision<0.22.1,>=0.21.0`.
#   The ROCm 7.2 index only ships torch 2.11.0/2.12.x/2.13.0 -- NO wheel satisfies the
#   pin. `pip install -e .` therefore resolves torch from PyPI and installs the CUDA
#   build plus the whole nvidia-* stack over your ROCm torch (confirmed with --dry-run:
#   "Collecting torch<2.7.1,>=2.7.0 ... torch-2.7.0-cp312-...whl" plus
#   "Collecting nvidia-cudnn-cu12==9.5.1.17"). The workaround is to install the
#   dependency closure WITHOUT the torch-family pins, then Composer and Foundry with
#   --no-deps. Do NOT edit upstream's setup.py; --no-deps is enough.
#
#   cd training/llm/llmfoundry
#   python3 -m venv .env_train_llm_llmfoundry
#   source .env_train_llm_llmfoundry/bin/activate
#   export HIP_VISIBLE_DEVICES=7 CUDA_VISIBLE_DEVICES=7    # never set these to ""
#   pip install -U pip setuptools wheel
#
#   # 1. ROCm torch FIRST (vision+audio too, so nothing pulls a CUDA wheel later)
#   pip install torch==2.11.0 torchvision==0.26.0 torchaudio==2.11.0 \
#     --index-url https://download.pytorch.org/whl/rocm7.2
#
#   # 2. constraints => an accidental torch swap fails loudly instead of silently
#   printf 'torch==2.11.0+rocm7.2\ntorchvision==0.26.0+rocm7.2\ntorchaudio==2.11.0+rocm7.2\n' \
#     > /tmp/rocm_constraints.txt
#
#   # 3. Composer deps MINUS torch/torchvision/torchaudio, then Composer itself
#   pip install -c /tmp/rocm_constraints.txt \
#     apache-libcloud coolname databricks-sdk google-cloud-storage importlib-metadata \
#     'mlflow>=2.14.1,<3.0' 'mosaicml-cli>=0.5.25,<0.8' 'numpy<2.3.0,>=1.21.5' oci \
#     'packaging<25.1,>=21.3.0' 'pillow<12,>=10.3.0' psutil py-cpuinfo 'pynvml<12,>=11.5.0' \
#     pyyaml requests tabulate==0.9.0 'torch_optimizer<0.4,>=0.3.0' \
#     'torchmetrics<1.7.5,>=1.0' tqdm 'wandb<0.19,>=0.13.2'
#   pip install --no-deps mosaicml==0.32.1
#
#   # 4. Foundry deps MINUS torch and MINUS mosaicml
#   pip install -c /tmp/rocm_constraints.txt \
#     'accelerate>=0.25,<1.9' 'transformers>=4.51.0,<4.52' 'mosaicml-streaming>=0.12.0,<0.13' \
#     'datasets>=3.3.2,<3.7' fsspec==2023.6.0 sentencepiece==0.2.0 einops==0.8.1 \
#     'omegaconf>=2.2.3,<3' 'slack-sdk<4' onnx==1.18.0 onnxruntime==1.22.0 'boto3>=1.21.45,<2' \
#     'huggingface-hub[hf_xet]>=0.30.0,<0.34' 'beautifulsoup4>=4.12.2,<5' 'tenacity>=8.2.3,<10' \
#     'catalogue>=2,<3' 'typer<1' GitPython==3.1.44 python-dotenv
#
#   # 5. the checkout, inside the git-ignored venv dir
#   cd .env_train_llm_llmfoundry
#   git clone --depth 1 https://github.com/mosaicml/llm-foundry.git
#   cd llm-foundry && pip install --no-deps -e .
#
#   NEVER `pip install flash-attn` on ROCm here -- there is no build for gfx950 and the
#   YAML's `attn_implementation: flash_attention_2` must be overridden with `sdpa`
#   instead (otherwise: "ValueError: use_flash_attention_2 is set to True, but
#   flash-attention 2 is not installed").
#   `pip check` will report 3 cosmetic metadata conflicts (llm-foundry/mosaicml want
#   torch<2.7.1, mosaicml wants torchvision<0.22.1). Nothing asserts on them at runtime.
#   If pip ever does replace torch, restore it with:
#     pip install --force-reinstall --no-deps torch==2.11.0 \
#       --index-url https://download.pytorch.org/whl/rocm7.2

# PATH C: AMD / ROCm (BETA, per the upstream README's "AMD (BETA support)" section)
#   NOTE: this is upstream's documented recipe, kept for reference. It does NOT work
#   as written on ROCm 7.x -- the rocm5.4.2 index is gone, and `pip install -e .`
#   BEFORE the ROCm torch line means pip installs CUDA torch and then the ROCm wheels
#   fight it. Use PATH C-ROCm7.2 above instead.
#   python3 -m venv llmfoundry-venv-amd && source llmfoundry-venv-amd/bin/activate
#   pip install cmake packaging torch
#   git clone https://github.com/mosaicml/llm-foundry.git
#   cd llm-foundry
#   pip install -e .               # NOT `.[gpu]` -- that extra is CUDA-only
#   pip3 install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/rocm5.4.2
#   # then the ROCm flash-attention fork (ROCmSoftwarePlatform/flash-attention),
#   # per the instructions upstream links. Upstream notes there is no Docker image
#   # where everything works and that package up/downgrades may be needed
#   # (they needed numpy==1.23.5). Blog evidence: mosaicml.com/blog/amd-mi250.
