# Environment for train_llm_openrlhf.py (Python 3.12), NVIDIA / CUDA 13.0.
# prepare_data_openrlhf.py is stdlib-only and needs nothing from this file.
#
# Install order:
#   pip install torch numpy            # torch FIRST -> the current CUDA 13 build
#   pip install -r requirements_openrlhf.txt
#   # The flash-attn wheel openrlhf[vllm] pulls is ABI-broken against torch 2.13.0, and
#   # openrlhf imports flash_attn at module load, so every trainer dies until it is rebuilt:
#   pip uninstall -y flash-attn
#   MAX_JOBS=32 TORCH_CUDA_ARCH_LIST="9.0" FLASH_ATTENTION_FORCE_BUILD=TRUE \
#     pip install flash-attn==2.8.3 --no-build-isolation --no-cache-dir   # needs nvcc
#   # DeepSpeed JIT-compiles FusedAdam at first run: `ninja` on PATH + CUDA_HOME set.
#
# Do not `pip install -r` this file on ROCm: openrlhf requires flash-attn (no ROCm wheel)
# and vllm==0.27.1 would overwrite ROCm torch with a CUDA build. Use the --no-deps venv
# route (SFT/DPO) or the rocm/verl container (PPO/GRPO) in readme_openrlhf.md, and run with
# --ds.attn_implementation sdpa and no --ds.packing_samples.

# Core. [vllm_latest] takes the newest vLLM instead of the pinned one;
# [vllm,ring,liger] adds ring-attention and Liger kernels.
openrlhf[vllm]

# Ray places the actor / critic / reference / reward models and the vLLM engines across
# GPUs, and provides the `ray job submit` entrypoint this script uses.
ray[default]>=2.40.0

# ZeRO sharding for the training-side models. Muon optimizer needs >= 0.18.2.
deepspeed>=0.18.2

# Read by train_llm_openrlhf.py to load dev.env (HF_TOKEN).
python-dotenv==1.2.2

# Packages that must not come from a plain `pip install`
#   flash-attn        always with --no-build-isolation, AFTER torch is present
#   ring_flash_attn   only for --ds.ring_attn_size; `ring` extra above, or
#                     pip install ring_flash_attn --no-build-isolation
#   torch             from the PyTorch index matching your CUDA, e.g.
#                     pip install torch --index-url https://download.pytorch.org/whl/cu126
#   OpenRLHF source   git clone https://github.com/OpenRLHF/OpenRLHF.git && pip install -e .
#
# Inside NVIDIA NGC pytorch containers, remove the colliding packages first:
#     pip uninstall xgboost transformer_engine flash_attn pynvml -y

# Optional
# Experiment tracking. wandb via --logger.wandb.key, tensorboard via --logger.tensorboard_dir.
wandb>=0.18.0
tensorboard>=2.18.0
