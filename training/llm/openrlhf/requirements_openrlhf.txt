# --- AMD / ROCm reality (tested on 1x MI355X gfx950, ROCm 7.2.4, py3.12) ------------
#
# `pip install -r` of THIS file fails on ROCm. Do not try. Root cause, in order:
#
#   1. openrlhf lists flash-attn==2.8.3 in install_requires -- NOT in an extra (true for
#      0.9.x .. 0.11.0). So `pip install openrlhf[vllm]` always tries to build flash-attn:
#          ERROR: Failed to build 'flash-attn' when getting requirements to build wheel
#          ModuleNotFoundError: No module named 'torch'
#      flash-attn 2.8.3 *does* allow gfx950 (setup.py allowed_archs) and bundles
#      composable_kernel, but it is a multi-hour hipcc build with no prebuilt ROCm wheel.
#
#   2. vllm==0.27.1 (what the [vllm] extra pins) is CUDA-only ON PyPI. It declares
#      torch==2.13.0 (the plain CUDA wheel), flashinfer-python, nvidia-cudnn-frontend and
#      nvidia-cutlass-dsl[cu13]. Installing it OVERWRITES a working ROCm torch. This blocks
#      PPO/GRPO *on the pip route only* -- ROCm builds of vLLM exist in AMD's first-party
#      images, and OpenRLHF drives one of them fine. See "ROCm CONTAINER ROUTE" at the
#      bottom of this file: PPO and GRPO both trained on 4x MI355X that way.
#
# WHICH PINS RESOLVE ON ROCm
#   resolve fine : ray[default]==2.55.0 (detects MI355X, accelerator_type:AMD-Instinct-MI355X-OAM)
#                  deepspeed==0.19.5 (pure-python wheel, JIT ops -- verified training on gfx950)
#                  transformers==5.15.0, accelerate, datasets, peft, optimum, torchmetrics,
#                  torchdata, wandb, tensorboard, python-dotenv, loralib, jsonlines
#   installs but  bitsandbytes (PyPI wheel is the CUDA build; AMD publish separate ROCm
#   is wrong-arch wheels), pynvml / nvidia-ml-py (NVIDIA-only; unused on the SFT/DPO path)
#   CUDA-ONLY    : flash-attn==2.8.3, vllm==0.27.1, ring_flash_attn
#
# WORKING INSTALL LINES -- VENV ROUTE (SFT + DPO only; for PPO/GRPO see the container
# route at the bottom of this file):
#
#   python3 -m venv .env_openrlhf && source .env_openrlhf/bin/activate
#   export HIP_VISIBLE_DEVICES=0 CUDA_VISIBLE_DEVICES=0      # never leave these empty
#
#   # ROCm torch FIRST, or a later resolve will pull a CUDA build.
#   pip install torch==2.11.0 --index-url https://download.pytorch.org/whl/rocm7.2
#
#   # --no-deps is what skips the unconditional flash-attn requirement.
#   pip install openrlhf==0.11.0 --no-deps
#
#   pip install accelerate aiohttp bitsandbytes datasets deepspeed==0.19.5 einops \
#     'grpcio>=1.74.0' 'huggingface-hub>=1.0.0' isort jsonlines loralib optimum \
#     'optree>=0.15.0' packaging peft pylatexenc 'pynvml>=12.0.0' 'ray[default]==2.55.0' \
#     sympy tensorboard torchdata torchmetrics tqdm transformers==5.15.0 \
#     transformers-stream-generator wandb wheel python-dotenv
#
#   # Then shim the pure-python half of flash-attn (bert_padding + utils/distributed, both
#   # plain torch+einops) into site-packages/flash_attn/ -- openrlhf/models/actor.py imports
#   # it at module level, so even SFT dies without it. Details in readme_openrlhf.md.
#
#   pip check   # expect only: openrlhf 0.11.0 requires flash-attn==2.8.3, which is not installed
#   python -c "import torch; print(torch.__version__)"   # must still end in +rocm7.2
#
# RUNTIME FLAGS THAT MUST CHANGE ON ROCm
#   --ds.attn_implementation sdpa      (flash_attention_2 is unavailable)
#   DROP --ds.packing_samples          (train_sft.py:324 silently forces flash_attention_2
#                                       back on whenever packing is enabled)
#   Both are ON by default in train_llm_openrlhf.py and cannot be disabled from the CLI --
#   edit the defaults, or call `deepspeed --module openrlhf.cli.train_sft` directly.
#
# Verified working on MI355X: torch 2.11.0+rocm7.2, deepspeed 0.19.5, transformers 5.15.0,
# ray 2.55.0, openrlhf 0.11.0 (--no-deps). SFT and DPO both step and checkpoint.

# --- ROCm CONTAINER ROUTE for the RL tiers (PPO / GRPO / RLOO / REINFORCE++) ----------
# TESTED on 4x MI355X (gfx950), ROCm 7.2.4 host. GRPO: 3 steps, exit 0, HF
# checkpoint written. PPO (with critic): 3 steps, exit 0, critic_loss 0.213 -> 0.088.
# Full walkthrough + evidence: readme_openrlhf.md, "RL tiers unblocked on MI355X".
#
# There is no pip line for this. The dependency you cannot get from PyPI -- a ROCm build of
# vLLM -- comes from the image; OpenRLHF is then dropped in with --no-deps.
#
#   IMAGE: rocm/verl:verl-0.7.1.amd0_rocm7.0.2_ubuntu22.04_py3.12_vllm0.20.2
#          (rocm/vllm works the same way; this one was chosen because it was already local)
#          ships torch 2.9.1.dev+rocm7.0.2 | vllm 0.20.2rc1.dev253+rocm702 | ray 2.55.1
#                transformers 5.14.1 | flash_attn 2.8.4 (ROCm/CK) | amd-aiter
#          -> flash-attn blocker (1) and vLLM blocker (2) are BOTH void inside it.
#
#   docker run -d --name orlhf_mi355x \
#     --device /dev/kfd --device /dev/dri/renderD128 --device /dev/dri/renderD136 \
#     --device /dev/dri/renderD144 --device /dev/dri/renderD152 \
#     --group-add 44 --group-add 993 --ipc=host --cap-add=SYS_PTRACE \
#     --security-opt seccomp=unconfined --shm-size=64g --network=host \
#     -e HF_HOME=/hf_cache -v "$PWD":/workspace/train_llm_openrlhf \
#     -v "$HF_HOME":/hf_cache -v "$OUTPUT_DIR/train_llm_openrlhf":/outputs \
#     -w /workspace/train_llm_openrlhf <IMAGE> sleep infinity
#
#   # THE ONLY pip line needed inside. --no-deps everywhere is load-bearing: a plain
#   # `pip install openrlhf` re-triggers the flash-attn source build AND the CUDA torch swap.
#   docker exec orlhf_mi355x pip install --no-deps \
#     openrlhf==0.11.0 deepspeed==0.19.5 hjson py-cpuinfo nvidia-ml-py pynvml python-dotenv
#   # (deepspeed + its 4 pure-python deps are the ONLY things the image was missing;
#   #  python-dotenv only if you drive it through train_llm_openrlhf.py)
#
#   docker exec orlhf_mi355x bash -lc \
#     'ulimit -n 262144; ray start --head --num-gpus 4 --num-cpus 64 --port 6379 --dashboard-port 8265'
#
# RUNTIME FLAGS ON THIS ROUTE
#   KEEP --ds.packing_samples and --ds.attn_implementation flash_attention_2 (both are the
#     launcher defaults and both WORK here -- ROCm flash-attn 2.8.4 is preinstalled)
#   --vllm.sync_backend nccl        resolves to RCCL; actor->engine weight sync works as-is
#   colocate_all wants  vllm.num_engines * vllm.tensor_parallel_size == actor GPUs
#   --reward.remote_url must be a path visible to the Ray WORKERS (it is imported inside the
#     vLLM engine actor) -- use a mounted absolute path, e.g. /outputs/reward_func.py
#   --working_dir must NOT be this folder: Ray would upload .env_openrlhf/
#     (libmagma.so 1.29 GB, libMIOpen.so 1.02 GB, ...). Point it at a dir holding just data/.
#
# GOTCHA THAT LOOKS LIKE A ROCm BUG BUT IS NOT: the image's nofile soft limit is 1024, so
# the raylet SIGABRTs with `epoll: Too many open files [system:24]` as soon as a driver
# connects. Run `ulimit -n 262144` in every shell that starts or submits to Ray.
