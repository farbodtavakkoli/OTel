# AMD Primus (Primus-LM) - large-model training on AMD Instinct GPUs (ROCm)
#
# IMPORTANT: this is NOT a normal pip environment. AMD publishes Docker images with
# the full ROCm training stack; the pip wheel installs the `primus-cli` launcher ONLY.
# Do not install this file into a CUDA venv and expect training to work.

#
# PATH A (recommended): container -- the verified route on ROCm.
#
#   docker pull rocm/primus:v26.5
#   # digest sha256:3040bf42974d791dd42de2e36b3c919a00869a5754cfc57a06b96d004c55eed1
#   # 54.8 GB on disk
#
#   docker run -d --name primus --device /dev/kfd --device /dev/dri \
#       --group-add video --group-add render --ipc=host \
#       --cap-add=SYS_PTRACE --security-opt seccomp=unconfined \
#       --shm-size 64G \
#       -v /path/to/hf_cache:/root/.cache/huggingface \
#       -v $PWD:/workspace/folder \
#       rocm/primus:v26.5 sleep infinity
#
# GPU SCOPING: primus-cli OVERWRITES HIP_VISIBLE_DEVICES in
# runner/helpers/envs/base_env.sh, so `-e HIP_VISIBLE_DEVICES=2,3` is silently
# ignored and you train on GPU 0,1. Scope with explicit devices instead:
#   --device /dev/dri/renderD<N> per GPU (see readme_primus.md section 2).
# Never set CUDA_VISIBLE_DEVICES alongside HIP_VISIBLE_DEVICES in this container:
# it aborts with "Conflicting visibility of agent-N".
#
# REQUIRED FIX inside the container: the image ships datasets 3.6.0, too old to
# parse current Hub dataset cards ("Feature type 'List' not found"):
#   docker exec <container> pip install "datasets==4.3.0"
#
# The whole stack ships pre-built at /workspace/Primus -- no pip install needed.
#
#   # Megatron-LM and TorchTitan backends:
#   docker pull rocm/primus:v26.5
#   # MaxText (JAX) backend:
#   docker pull rocm/jax-training:maxtext-v26.5
#
#   git clone --recurse-submodules https://github.com/AMD-AGI/Primus.git
#   cd Primus
#   git checkout release/v26.5
#   git submodule update --init --recursive
#
# Check current tags: https://hub.docker.com/r/rocm/primus/tags

#
# PATH B: pip wheel (CLI only; still launches training inside the container)
#   python -m venv primus-env && source primus-env/bin/activate
#
#   # from the GitHub Pages index (pin the version):
#   pip install "primus==26.5.0" --no-deps \
#       --extra-index-url https://amd-agi.github.io/Primus/simple/
#
#   # or straight from git, pinned to a tag:
#   pip install "git+https://github.com/AMD-AGI/Primus.git@v26.2.0rc1"
#
#   # fetch the pinned backend sources (incl. nested submodules):
#   primus-cli deps sync --dir ~/.cache/Primus/third_party
#
# `primus-cli direct` auto-runs `deps sync` on first use when sources are
# missing; set PRIMUS_AUTO_DEPS_SYNC=0 to disable that.

#
# PATH C: bare metal host
# See docs/01-getting-started/installation.md -> "Bare-metal host setup".

#
# Prerequisites (host, all paths)
#   ROCm drivers >= 7.0
#   Docker >= 24.0 with ROCm support
#   AMD Instinct GPU: MI300X / MI325X (gfx942) or MI350X / MI355X (gfx950)
#   Verify with: rocm-smi && docker --version

# Extra deps for THIS folder's launcher only (safe in any venv)
