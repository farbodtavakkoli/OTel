# torchtune - continued pre-training example (LoRA, text-completion dataset)
#
#   python3.12 -m venv .venv && source .venv/bin/activate
#   pip install -r requirements_torchtune.txt
#
# Then download the base weights once:
#   tune download Qwen/Qwen2.5-7B --output-dir ./assets/Qwen2.5-7B
#
# Install torch first, matched to the accelerator, then the rest of this file.
#
# AMD / ROCm 7.2 — no torch nightly and no --no-deps on the torchtune install:
#   pip install --index-url https://download.pytorch.org/whl/rocm7.2 torch torchvision
#   pip install torchtune==0.6.1 python-dotenv
#
# NVIDIA / CUDA 13 — the default stable torch wheel is already native cu130, so no
# --index-url is needed:
#   pip install torch numpy                       # -> the current CUDA 13 build
#   pip install torchtune==0.6.1 torchvision python-dotenv
#   pip install torchao==0.10.0 --no-deps         # Required
#
# Why torchao==0.10.0 (both platforms): torchtune 0.6.1 does
# `import torchao.dtypes.nf4tensor` at import time and does not pull torchao itself.
# A bare `pip install torchao` grabs 0.18.x, which moved that module, so
# `import torchtune` dies with
# ModuleNotFoundError: No module named 'torchao.dtypes.nf4tensor'.
# --no-deps keeps the pin from dragging torch off its accelerator build.
#
# Do not pip install flash-attn on either platform — torchtune uses torch flex/SDPA
# attention and exposes no attn_implementation knob in these configs.
# Keep `device: cuda` in the YAML on both: ROCm torch exposes HIP via torch.cuda.

torchtune==0.6.1
torch>=2.5.0            # H100: resolves to 2.13.0+cu130 from the default index (no --index-url)
torchvision
torchao==0.10.0         # Pinned: torchtune 0.6.1 needs the 0.6-era torchao.dtypes.nf4tensor API

# this folder's launcher / converter
python-dotenv>=1.0.1
