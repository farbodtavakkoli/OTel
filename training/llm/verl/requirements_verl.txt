# verl - RL post-training (GRPO / PPO / DAPO / RLOO) for LLMs
#
# verl orchestrates THREE stacks at once: a training engine (FSDP2 or Megatron),
# a rollout/inference engine (vLLM or SGLang), and Ray for placement. Those three
# have tight, conflicting version constraints, which is why upstream ships
# Docker images and a uv.lock. Prefer the image; treat this file as the venv
# fallback.
#
# RECOMMENDED instead of pip: the official images
#   NVIDIA     : verlai/verl:uv.cu130   # 43.3 GB on disk. NO ready .venv: ships only the
#                # uv package cache. After `docker run`, inside the container run
#                #   cd /workspace/verl && python3 manage_envs.py sync vllm -- --frozen
#                # flash_attn is NOT installed in the vllm slice; force
#                #   +actor_rollout_ref.model.override_config.attn_implementation=sdpa   (see readme).
#                # verl 0.9 reward drift: also pass reward.custom_reward_function.{path,name} (abs path).
#                # `unset HTTP_PROXY HTTPS_PROXY ALL_PROXY http_proxy https_proxy all_proxy` before pulling.
#                # Query live tags: curl -s "https://hub.docker.com/v2/repositories/verlai/verl/tags/?page_size=100&ordering=last_updated"
#   AMD (ROCm) : rocm/verl:verl-0.7.1.amd0_rocm7.0.2_ubuntu22.04_py3.12_vllm0.20.2
#                (source build: docker/rocm/Dockerfile.rocm upstream — see
#                 docs/amd_tutorial/amd_quick_start.rst)
# Check https://verl.readthedocs.io/en/latest/start/install.html for the current tag.
#
#
# AMD MI355X (gfx950) / ROCm 7.2 — verified. Details in readme_verl.md.
#
# Use the container. There is NO working pip route to a trainable state on ROCm: no ROCm
# vLLM wheel is published on PyPI and modern verl has no vLLM-free rollout engine
# (`rollout.name=hf` was removed in 0.6.0).
#
#   docker pull rocm/verl:verl-0.7.1.amd0_rocm7.0.2_ubuntu22.04_py3.12_vllm0.20.2
#   # 42.5 GB on disk. The tag understates the contents: it ships verl 0.8.0.dev0 with
#   # flash_attn preinstalled (ROCm/CK build). Runs against a ROCm 7.2.4 host driver.
#
# Required at runtime on ROCm (see readme for why):
#   RAY_ACCEL_ENV_VAR_OVERRIDE_ON_ZERO=0   # else Triton "0 active drivers" in TaskRunner
#   # set NEITHER HIP_VISIBLE_DEVICES NOR CUDA_VISIBLE_DEVICES — vLLM's ROCm
#   # platform raises if both are set and differ; pin GPUs with per-device
#   # `--device /dev/dri/renderDxxx` on `docker run` instead.
#   VLLM_ROCM_USE_AITER=1                  # image default, left on
#   VLLM_ROCM_USE_AITER_MOE=0              # AITER MoE kernels are unsafe on gfx950
#
# The pip lines below DO work on ROCm but only get you data prep + `--dry-run`
# (converter, launcher, Hydra override validation) — not training:
#   python3 -m venv .env_verl && source .env_verl/bin/activate
#   pip install torch==2.11.0 --index-url https://download.pytorch.org/whl/rocm7.2
#   git clone https://github.com/volcengine/verl.git && cd verl
#   pip install --no-deps -e .
#   pip install accelerate codetiming datasets dill hydra-core "numpy<2.0.0" \
#       pandas peft "pyarrow>=19.0.0" pybind11 pylatexenc "ray[default]>=2.41.0" \
#       torchdata "tensordict>=0.8.0,<=0.10.0,!=0.9.0" transformers wandb \
#       "packaging>=20.0" tensorboard python-dotenv
#   # NEVER `pip install flash-attn` on ROCm — the rocm/verl image already ships a
#   # ROCm/CK build, and the CUDA wheel cannot build.
#

# Install verl itself from source (recommended over the PyPI wheel; the
# examples/, recipe/ and config defaults only exist in the git tree):
#   git clone https://github.com/volcengine/verl.git
#   cd verl
#   pip install --no-deps -e .
#
# Pin a release instead of main if you want reproducibility:
#   pip install "verl[vllm]==0.5.0"
