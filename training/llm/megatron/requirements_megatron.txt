# Pure-Python extras for train_llm_megatron.py (Megatron-LM continued pre-training).
# Megatron-LM itself is a SOURCE install (pretrain_gpt.py and tools/ exist only in the
# git tree), and torch / transformer-engine / apex / flash-attn are not listed here on
# purpose — take them from a container, never from a plain `pip install`.
#
# NVIDIA route — NGC PyTorch container (recommended; TE + Apex + flash-attn pre-built):
#   unset HTTP_PROXY HTTPS_PROXY ALL_PROXY http_proxy https_proxy all_proxy  # nvcr.io 403s behind a proxy
#   docker run --gpus all --ipc=host --ulimit memlock=-1 --ulimit stack=67108864 \
#     -v $PWD:/workspace -v /data:/data -it nvcr.io/nvidia/pytorch:25.06-py3 bash
#   # inside, in this order:
#   cd /opt && git clone --depth 1 https://github.com/NVIDIA/Megatron-LM.git
#   pip install --no-build-isolation -e Megatron-LM   # reuses the container's torch/TE/Apex
#   pip install "nvidia-resiliency-ext>=0.6.0"        # image ships 0.4.0; below Megatron's
#                                                     # minimum -> `import megatron.core` dies
#                                                     # with AttributeError: no __version__
#   pip install -r requirements_megatron.txt          # transformers/dotenv/nltk not in image
#   # Outside a container, TE must be built against your torch:
#   #   pip install --no-build-isolation transformer-engine[pytorch]
#
# AMD / ROCm 7.2 route (MI355X, gfx950) — venv + source clone, no container. AMD also
# publishes rocm/megatron-lm images (https://hub.docker.com/r/rocm/megatron-lm/tags).
#   python3 -m venv .env_megatron && source .env_megatron/bin/activate  # torchrun on PATH
#   pip install --index-url https://download.pytorch.org/whl/rocm7.2 torch==2.11.0  # FIRST
#   pip install pybind11 "packaging>=24.2" numpy   # BEFORE the editable install, or
#                                                  # --no-build-isolation fails on metadata
#   git clone --depth 1 https://github.com/NVIDIA/Megatron-LM.git .env_megatron/Megatron-LM
#   pip install --no-build-isolation -e .env_megatron/Megatron-LM
#   pip install -r requirements_megatron.txt
#
# Do not install transformer-engine, apex or flash-attn on ROCm (CUDA-only). Pass these
# launcher overrides instead — no TOML edit needed:
#   --set transformer_impl='"local"'           # TOML ships "transformer_engine"
#   --set no_rope_fusion=true                  # else: apply_rope_fusion needs TE >= 1.4
#   --set no_persist_layer_norm=true           # else: not supported by torch LayerNorm
#   --set no_gradient_accumulation_fusion=true # else: fused_weight_gradient_mlp_cuda missing
# And sequence_parallel must stay false: torch LayerNorm cannot do SP without TE.
