# Red Hat training-hub GRPO stack. Install with ./setup.sh.
#
# `pip install training-hub[grpo]` is deliberately NOT used: its extras pin torch/vLLM/
# transformers and would replace the stack vLLM installs.

# Same tensordict constraint as the verl stack: on 0.14.x verl's tu.get() returns a list for the
# per-rank `metrics` dict and ray_trainer.py's ["mfu"] lookup raises TypeError.
tensordict==0.13.0

# verl's ray_trainer imports torchdata.stateful_dataloader at module level; it is not a vLLM dep.
torchdata==0.11.0

# LoRA, dataset and device-map plumbing (not transitive deps of vLLM).
peft
datasets
accelerate

# verl + training_hub runtime deps their --no-deps installs skip.
ray[default]
hydra-core
codetiming
pylatexenc
math_verify
latex2sympy2_extended
uvicorn
fastapi
pybind11
instructlab-training
rhai-innovation-mini-trainer
deprecated

# instructlab-training pulls trl transitively, and verl's monkey patch imports
# AutoModelForCausalLMWithValueHead whenever trl is importable at all: guarded on the package being
# present, not on the symbol. trl 1.13 dropped it from both `trl` and `trl.experimental.ppo`, so the
# actor worker fails with ImportError while building the model. 1.12.0 still carries it.
trl==1.12.0
