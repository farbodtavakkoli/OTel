# Environment for train_llm_torchtitan.py (torchtitan continued pre-training / SFT).
# Install into a DEDICATED venv - torchtitan tracks PyTorch nightly and conflicts
# with the pinned torch in the other trainers in this repo.
#
#
# torchtitan is a SOURCE install (the Python-config surface and the conversion scripts
# only exist in the git tree). Do not also `pip install torchtitan` from PyPI into the
# same venv — the stable wheel pins its own torch/torchao and shadows the clone.
#
# Install order, NVIDIA (CUDA 13) — see readme_torchtitan.md section 7.3:
#   python3.12 -m venv .env_torchtitan && source .env_torchtitan/bin/activate
#   export CUDA_VISIBLE_DEVICES=0        # NO HIP_VISIBLE_DEVICES on NVIDIA
#   # 1. NIGHTLY torch FIRST (proxy must be unset if download.pytorch.org is blocked)
#   pip install --pre torch --index-url https://download.pytorch.org/whl/nightly/cu130
#   # 2. torchtitan from source, PINNED (a fresh HEAD breaks recipe_torchtitan.py's
#   #    CheckpointManager import)
#   git clone https://github.com/pytorch/torchtitan .env_torchtitan/torchtitan_src
#   git -C .env_torchtitan/torchtitan_src checkout 03be241
#   pip install -r .env_torchtitan/torchtitan_src/requirements.txt
#   pip install torchdata
#   # 3. this file
#   pip install -r requirements_torchtitan.txt
#
# AMD / ROCm 7.2 (MI355X, gfx950) — same sequence with the ROCm nightly index; see
# readme_torchtitan.md section 7.1:
#   export HIP_VISIBLE_DEVICES=0,1 CUDA_VISIBLE_DEVICES=0,1   # never set these to ""
#   pip install --pre torch --index-url https://download.pytorch.org/whl/nightly/rocm7.2
#   pip install --pre torchdata --index-url https://download.pytorch.org/whl/nightly/cpu
#
# The nightly is required on both vendors: against torchtitan main a stable torch
# 2.13.0 dies in fully_shard on ROCm (dp_mesh_dims/DTensor) and at set_pg_timeouts on
# CUDA (torch.distributed has no attribute 'set_timeout'). Stable wheels are only valid
# with an older, matching torchtitan release. Do not install flash-attn on either vendor
# — torchtitan uses SDPA / FlexAttention. If a later pip step swaps torch, reinstall it
# from the right nightly index (check torch.version.hip).
