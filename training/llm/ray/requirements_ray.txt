# Environment for train_llm_ray.py (Ray Train orchestration of an HF/TRL SFT fine-tune).
# Python 3.10-3.12, in a dedicated venv:
#
#   python3.12 -m venv ~/.venv-ray && source ~/.venv-ray/bin/activate
#   pip install --upgrade pip
#   pip install -r requirements_ray.txt
#
# CUDA 13: the torch==2.11.0 pin below is a CUDA 13 build on PyPI, so no --index-url is
# needed. Install torch first, then the rest without re-resolving it:
#   pip install torch==2.11.0 numpy            # CUDA 13 build, straight from PyPI
#   pip install ray[train]==2.57.0 transformers==5.5.0 trl==0.27.0 peft==0.18.1 \
#               datasets==4.3.0 accelerate==1.14.0 python-dotenv 'pyarrow>=23.0.1'
#   python -c "import torch;print(torch.__version__, torch.version.cuda)"   # 2.11.0 13.0
#   export CUDA_VISIBLE_DEVICES=<your gpus>    # do not set RAY_EXPERIMENTAL_NOSET_CUDA_VISIBLE_DEVICES
#                                              # (ROCm-only, unneeded here)
#
# ROCm 7.2: do not `pip install -r` this file directly -- install the ROCm torch wheel first so
# pip never resolves the CUDA one. Every other pin is unchanged:
#   pip install --index-url https://download.pytorch.org/whl/rocm7.2 torch==2.11.0
#   pip install ray[train]==2.57.0 transformers==5.5.0 trl==0.27.0 peft==0.18.1 \
#               datasets==4.3.0 accelerate==1.14.0 python-dotenv 'pyarrow>=23.0.1'
#   export HIP_VISIBLE_DEVICES=0,1          # Ray scopes AMD GPUs with HIP_VISIBLE_DEVICES,
#                                           # NOT ROCR_VISIBLE_DEVICES.
#   export RAY_EXPERIMENTAL_NOSET_HIP_VISIBLE_DEVICES=1
#                                           # Required for --num_workers > 1; without it every
#                                           # rank but 0 dies with "invalid device ordinal".
#                                           # See readme_ray.md 8.1.
#
# Do not install flash-attn on either platform: the script hard-codes
# attn_implementation="sdpa" and exposes no flag to switch it.
# Every node in the cluster must have an identical environment -- same Ray version, same Python
# minor version, same torch build -- or Ray refuses the worker.

# Ray Train (the orchestrator). 2.43+ is required for the Train V2 API used by this script;
# V2 is default-on from ~2.57, on 2.43-2.56 also export RAY_TRAIN_V2_ENABLED=1.
ray[train]==2.57.0

# Training stack (runs inside the Ray worker)
torch==2.11.0                  # CUDA 13 build on PyPI; see headers for install order & ROCm
transformers==5.5.0
trl==0.27.0                    # SFTTrainer / SFTConfig
peft==0.18.1                   # LoRA adapters (--use_lora)
datasets==4.3.0
accelerate==1.14.0

# Utilities
python-dotenv                  # loads dev.env for HF_TOKEN
pyarrow>=23.0.1                # Ray Train's storage layer (pyarrow.fs); also a CVE floor

# OPTIONAL: cloud checkpoint storage. Multi-node runs cannot use head-node local disk;
# s3:// works through pyarrow's bundled S3FileSystem, boto3 only helps with credentials.
# boto3>=1.34

# OPTIONAL: experiment tracking

# wandb>=0.28.0
