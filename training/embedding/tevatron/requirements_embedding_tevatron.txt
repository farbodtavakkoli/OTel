# Environment for train_embedding_tevatron.py (Tevatron dense-retriever training).
# These are NOT preinstalled in the base training venv; install into a dedicated venv.
# If the root filesystem is tight, put the venv outside the repo on a larger volume
# ($DATA_DIR below is that volume):
#   python3 -m venv $DATA_DIR/envs/.env_tevatron
#
# AMD/ROCm: install the ROCm torch wheel FIRST, then this file, then the two git
# packages below (verified on MI355X gfx950 / ROCm 7.2.4 — installs as
# torch 2.11.0+rocm7.2, which satisfies the ==2.11.0 pin):
#   export PIP_CACHE_DIR=$DATA_DIR/pip_cache
#   pip install torch==2.11.0 --index-url https://download.pytorch.org/whl/rocm7.2
#   pip install -r requirements_embedding_tevatron.txt
#   pip install git+https://github.com/luyug/GradCache.git
#   pip install --no-deps git+https://github.com/texttron/tevatron.git
#   python -c "import torch; assert torch.version.hip and torch.version.cuda is None"
# If pip swapped in the CUDA wheel:
#   pip install --force-reinstall --no-deps torch==2.11.0 \
#     --index-url https://download.pytorch.org/whl/rocm7.2
#
# NVIDIA (CUDA): same order without the ROCm index — `pip install torch==2.11.0`
# from PyPI, then this file, then the two git packages.
# VERIFIED on H100 80GB (Hopper cc9.0), CUDA 13.0, driver 580.173.02, Python 3.12.3:
# the PyPI 2.11.0 wheel is already a native CUDA-13 build (installs as torch 2.11.0+cu130 with
# nvidia-*-cu13 deps) — NO --index-url needed. Re-verify after the git installs (they leave it intact):
#   pip install torch==2.11.0 numpy
#   pip install -r requirements_embedding_tevatron.txt
#   git clone https://github.com/luyug/GradCache.git   && pip install ./GradCache
#   git clone https://github.com/texttron/tevatron.git && pip install --no-deps ./tevatron   # @ dd06310
#   python -c "import torch; assert torch.version.cuda and torch.version.hip is None; print(torch.__version__)"
#   # -> 2.11.0+cu130

#   pip install flash-attn==2.8.3     # CUDA only; never install this on ROCm
