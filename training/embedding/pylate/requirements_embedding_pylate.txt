# Environment for train_embedding_pylate.py (PyLate ColBERT late-interaction fine-tuning).
# These are NOT preinstalled in the base training venv; install into a dedicated venv.
#
# NOTE: if the root filesystem is tight, put the venv outside the repo on a larger volume
# ($DATA_DIR below is that volume):
#   python3 -m venv $DATA_DIR/envs/.env_pylate
#   source $DATA_DIR/envs/.env_pylate/bin/activate
#
# AMD/ROCm: install the ROCm torch wheel FIRST, then this file:
#   export PIP_CACHE_DIR=$DATA_DIR/pip_cache
#   pip install torch==2.11.0 --index-url https://download.pytorch.org/whl/rocm7.2
#   pip install -r requirements_embedding_pylate.txt
#   python -c "import torch; print(torch.__version__, torch.version.hip, torch.version.cuda)"
#   # expected: 2.11.0+rocm7.2 7.2.26015 None
# Verified on MI355X (gfx950) / ROCm 7.2.4: installing pylate did NOT pull a CUDA torch —
# none of its dependencies pin torch. If it ever does, recover with:
#   pip install --force-reinstall --no-deps torch==2.11.0 \
#     --index-url https://download.pytorch.org/whl/rocm7.2
# Do NOT install flash-attn on ROCm (CUDA-only build) — the script auto-selects "sdpa".
#
# NVIDIA (CUDA): drop the --index-url line above; the default PyPI torch wheels ship CUDA.
#   pip install -r requirements_embedding_pylate.txt
# Optional CUDA-only extras: flash-attn (--attn_implementation flash_attention_2) and the
# fused MaxSim kernels, pylate[flash-maxsim] / pylate[lik] (--scores_backend flash|lik).
#
# VERIFIED on NVIDIA H100 80GB / CUDA 13.0 / driver 580.173.02:
#   export PIP_CACHE_DIR=$DATA_DIR/pip_cache
#   pip install "torch==2.11.0" numpy          # the torch==2.11.0 pin below IS honorable on CUDA:
#                                              #   a torch==2.11.0+cu130 wheel exists on default PyPI
#                                              #   (NO --index-url). A bare `pip install torch` would
#                                              #   instead give 2.13.0+cu130; either runs on driver 580.
#   pip install -r requirements_embedding_pylate.txt
#   python -c "import torch; print(torch.__version__, torch.version.cuda, torch.version.hip)"
#   # expected: 2.11.0+cu130 13.0 None
# As on MI355X, installing pylate did NOT pull/clobber torch (no dep pins it) — re-verify anyway;
# recover if it ever does with:
#   pip install --force-reinstall --no-deps "torch==2.11.0"   # default PyPI cu130 wheel
