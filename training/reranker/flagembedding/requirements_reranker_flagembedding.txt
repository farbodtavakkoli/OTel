# Environment for train_reranker_flagembedding.py (BGE / LLM reranker fine-tuning via FlagEmbedding).
# Install into a dedicated venv.
#
# AMD/ROCm 7.2 — install the ROCm torch wheel FIRST, then this file:
#   python3 -m venv $DATA_DIR/envs/.env_flagembedding
#   source $DATA_DIR/envs/.env_flagembedding/bin/activate
#   pip install torch==2.11.0 --index-url https://download.pytorch.org/whl/rocm7.2
#   pip install -r requirements_reranker_flagembedding.txt
#
# NVIDIA/CUDA 13 — no --index-url needed; torch==2.11.0 resolves to a native cu130 wheel
# on plain PyPI. Install torch first, then this file:
#   pip install torch==2.11.0 numpy               # -> 2.11.0+cu130
#   pip install -r requirements_reranker_flagembedding.txt
#   python -c "import torch;print(torch.__version__,torch.version.cuda)"  # re-verify
#   # optional, needs the CUDA toolkit:
#   # pip install flash-attn==2.8.3
# Do not install flash-attn on ROCm (CUDA-only build here) — utils.resolve_attn()
# keeps FlagEmbedding's --use_flash_attn False and the model falls back to sdpa.

# Core framework (encoder cross-encoders and decoder-only LLM rerankers)
FlagEmbedding==1.4.0

# Training stack.
# transformers is pinned <5 deliberately: FlagEmbedding 1.4.0's reranker data pipeline
# calls tokenizer.prepare_for_model(), which transformers 5.x removed, so the reranker
# path dies on the first batch under 5.x.
torch==2.11.0            # ROCm 7.2 pin. The same pin resolves to a cu130 wheel on PyPI (see header).
transformers==4.57.1     # <5 required on both ROCm and CUDA (FlagEmbedding needs tokenizer.prepare_for_model, removed in 5.x).
huggingface-hub==0.36.2
tokenizers==0.22.2
accelerate==1.14.0
datasets==5.0.1
peft==0.20.0
sentence-transformers==5.7.0   # 6.0.0 requires transformers>=5; keep 5.7.0 with the pin above
sentencepiece==0.2.2
protobuf==7.35.1
ir_datasets==0.6.3

# Utilities
python-dotenv==1.2.2
tensorboard==2.21.0
numpy==2.5.2
